    # pytest.main(["-x", "mytestdir"])
    # multiple calls to pytest.main() from the same process
    # (in order to re-run tests, for example) is not recommended.

-r flag
    default is equivalent to -rfE
    accepts characters after it, meaning “all except passes”.
        f - failed
        E - error
        s - skipped
        x - xfailed
        X - xpassed
        p - passed
        P - passed with output
    Special characters for (de)selection of groups:
        a - all except pP
        A - all
        N - none, display nothing (since fE is the default).

On any failure the exception information is stored on:
    sys.last_value, sys.last_type and sys.last_traceback.

When running Terraform in automation, the focus is usually on the core plan/apply cycle.
The main path, then, is broadly the same as for CLI usage:

Initialize the Terraform working directory.
Produce a plan for changing resources to match the current configuration.
Have a human operator review that plan, to ensure it is acceptable.
Apply the changes described by the plan.

Steps 1, 2 and 4 can be carried out using the familiar Terraform CLI commands, with some additional options:

terraform init -input=false to initialize the working directory.
terraform plan -out=tfplan -input=false to create a plan and save it to the local file tfplan.
terraform apply -input=false tfplan to apply the plan stored in the file tfplan.

def unarchive_asset2(
    asset_id: str, file_name: str, path: str = PORTAL_SUBMISSIONS_S3_BASE_PATH
) -> tuple:
    """
    Delete file format Archived status for an asset.
    asset_id -- Id of the asset in iconik.
    file_name -- filename for the asset in iconik.
    path -- Directory where file must be located in storage.
    """

    try:
        format_id = create_format(asset_id)
        file_set_id = create_file_set(asset_id, format_id, file_name, path)
        file_id = create_file(asset_id, format_id, file_set_id, file_name, path)

        # Get all asset's format for the asset_id
        url = "{base_url}/files/v1/assets/{asset_id}/formats/".format(
            base_url=ICONIK_BASE_URL, asset_id=asset_id
        )
        r = requests.get(url, headers=HEADERS, timeout=TIMEOUT_SECONDS)
        result = json.loads(r.text)

        # Delete all 'Archived' formats.
        for object in result["objects"]:
            # archive_status is None if it is Archived OR 'NOT_ARCHIVED' if not.
            if object["archive_status"]:
                continue
            # Archived format id .
            format_id = object["id"]
            url = "{base_url}/files/v1/assets/{asset_id}/formats/{format_id}/".format(
                base_url=ICONIK_BASE_URL, asset_id=asset_id, format_id=format_id
            )
            r = requests.delete(url, headers=HEADERS, timeout=TIMEOUT_SECONDS)

        return asset_id, file_id
    except Exception as ex:
        print(ex)
        raise
